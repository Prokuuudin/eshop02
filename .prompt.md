## Code Quality Standards for Phase 1

### TypeScript
```tsx
// ❌ NO implicit any
export const handler = (data) => { }
const { email } = formData  // eslint-disable-line should not appear

// ✅ YES explicit types
export const handler = (data: FormData): Promise<Response> => { }
const { email } = formData as { email: string }
```

### RSC Patterns
```tsx
// ✅ Correct: RSC + minimal client island
export async function Page() {
  const data = await fetchData()  // server-side
  return (
    <>
      <ServerComponent data={data} />
      <ClientForm />  {/* only interactive parts */}
    </>
  )
}

// ❌ Wrong: everything client-side
'use client'
export function Page() {
  const [data, setData] = useState()
  useEffect(() => {
    fetch('/api/data').then(r => r.json()).then(setData)
  }, [])  // CSR, no SEO
}
```

### Schema Validation
```tsx
// Test schema with: https://validator.schema.org
// Before commit, verify:
// 1. "@context": "https://schema.org"
// 2. "@type": one of (Article, Product, Organization, etc.)
// 3. All required fields present per schema.org docs
// 4. No hardcoded localhost URLs

console.log(JSON.stringify(schema))  // copy to validator
```
